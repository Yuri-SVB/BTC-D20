% Shared preamble for all language editions of the instructions sheet.
% Language files load this first, then their babel option, then the content.
\usepackage[a4paper, margin=1cm]{geometry}

% Per-edition page layout, called right after this file is input.
% Standard editions are a strict one-pager (table on the verso): margins as
% generous as the longest translation allows. Extended editions have no
% one-page commitment: roomier margins, standard two-page flow.
\newcommand{\standardlayout}{\geometry{hmargin=1.3cm, vmargin=1.1cm}}
\newcommand{\extendedlayout}{\geometry{hmargin=1.7cm, vmargin=1.5cm}}
\usepackage{enumitem}
\usepackage{xcolor}
\usepackage{tcolorbox}
\usepackage{qrcode}
\usepackage{fontawesome5}
\usepackage{titlesec}
\pagestyle{empty}

% Compact vertical rhythm so each edition fits a single A4 recto.
\titlespacing*{\section}{0pt}{0pt}{4pt}
\titlespacing*{\subsection}{0pt}{5pt}{2pt}
\setlist{itemsep=1pt, topsep=2pt, parsep=0pt}
\tcbset{boxsep=1pt, top=2pt, bottom=2pt, left=4pt, right=4pt,
        before skip=6pt, after skip=6pt}

% Box styles used by every edition.
\newtcolorbox{contextbox}[1]{colback=red!4, colframe=red!55!black, title={#1}}
\newtcolorbox{examplebox}[1]{colback=yellow!15, colframe=yellow!65!black, title={#1}}
\newtcolorbox{warnbox}[1]{colback=yellow!10, colframe=orange!90, title={#1}}
\newtcolorbox{infobox}[1]{colback=blue!5, colframe=blue!50, title={#1}}

% Footer line naming the sibling editions.
\newcommand{\editionsline}[1]{%
  {\centering\scriptsize\color{black!60}#1 --- EN \textperiodcentered{} ES
  \textperiodcentered{} IT \textperiodcentered{} PT-BR\par}}
% One-pagers pin it to the page bottom; kept glue-cheap (no center
% environment, zero-minimum fill) so it never pushes the page over.
\newcommand{\editionsfooter}[1]{\par\vspace*{\fill}\nopagebreak\editionsline{#1}}
% Multi-page editions let it follow the content instead.
\newcommand{\editionsfooterflow}[1]{\par\vspace{1.5em}\editionsline{#1}}
