% Integer division (quotient)
\newcommand{\intdiv}[2]{%
  \fpeval{floor(#1 / #2)}%
}

% Remainder (modulo)
\newcommand{\intmod}[2]{%
  \fpeval{#1 - #2 * floor(#1 / #2)}%
}

\newcommand{\nextcellarithm}{%
  \xdef\cellnum{\fpeval{\cellnum + 1}}%
  \csname cell\cellnum\endcsname%
  \xdef\paritya{\intmod{\cellnum}{2}}
  \xdef\parityb{\intmod{\intdiv{\cellnum}{16}}{2}}
  \xdef\parityc{\intmod{\intdiv{\cellnum}{256}}{2}}
}

\newcommand{\prevcellarithm}{%
  \xdef\cellnum{\fpeval{\cellnum - 1}}%
  \csname cell\cellnum\endcsname%
  \xdef\paritya{\intmod{\cellnum}{2}}
  \xdef\parityb{\intmod{\intdiv{\cellnum}{16}}{2}}
  \xdef\parityc{\intmod{\intdiv{\cellnum}{256}}{2}}
}
